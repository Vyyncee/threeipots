    print(f"\n{'='*80}")
    print(f"Dataset: {nom}")
    print('='*80)
    
    colonnes_categorielles = df.select_dtypes(include=['object', 'category']).columns.tolist()
    
    if colonnes_categorielles:
        print(f"\nColonnes catégorielles trouvées: {len(colonnes_categorielles)}")
        
        for col in colonnes_categorielles:
            nb_valeurs_uniques = df[col].nunique()
            nb_lignes = len(df)
            
            # Calcul de l'espace mémoire estimé pour One-Hot Encoding
            # Approximation: nb_lignes * nb_valeurs_uniques * 8 bytes
            memoire_estimee_mb = (nb_lignes * nb_valeurs_uniques * 8) / (1024 * 1024)
            
            print(f"\n  📊 Colonne '{col}':")
            print(f"      - Valeurs uniques: {nb_valeurs_uniques:,}")
            print(f"      - % de cardinalité: {(nb_valeurs_uniques/nb_lignes)*100:.2f}%")
            print(f"      - Mémoire estimée (One-Hot): ~{memoire_estimee_mb:.2f} MB")
            
            # Alertes
            if nb_valeurs_uniques > 1000:
                print(f"      ⚠️  ATTENTION: Trop de valeurs ! Risque de MemoryError")
            elif nb_valeurs_uniques > 100:
                print(f"      ⚠️  Attention: Beaucoup de valeurs")
            
            # Exemples de valeurs
            print(f"      - Exemples: {df[col].unique()[:3].tolist()}")
    else:
        print("  ✓ Aucune colonne catégorielle détectée")

# Vérification que les datasets soit compatibles pour la fusion

# ============================================================================
# CONFIGURATION : Choisir les datasets à comparer

# 1. Vérification des colonnes présentes dans chaque dataset
print("="*80)
print("COLONNES PRÉSENTES DANS CHAQUE DATASET")
print("="*80)
for nom, df in datasets_a_comparer.items():
    print(f"\nDataset {nom} ({len(df.columns)} colonnes):")
    print(df.columns.tolist())

# 2. Créer des sets de colonnes pour chaque dataset
colonnes_par_dataset = {nom: set(df.columns) for nom, df in datasets_a_comparer.items()}
noms_datasets = list(datasets_a_comparer.keys())

# 3. Trouver toutes les colonnes uniques (union de toutes les colonnes)
toutes_les_colonnes = set()
for colonnes in colonnes_par_dataset.values():
    toutes_les_colonnes.update(colonnes)

# 4. Analyser les problèmes de compatibilité
print("\n" + "="*80)
print("COLONNES AVEC PROBLÈMES DE COMPATIBILITÉ")
print("="*80)

problemes_trouves = False

for col in sorted(toutes_les_colonnes):
    # Vérifier dans quels datasets la colonne existe
    datasets_avec_colonne = [nom for nom, colonnes in colonnes_par_dataset.items() if col in colonnes]
    datasets_sans_colonne = [nom for nom in noms_datasets if nom not in datasets_avec_colonne]
    
    # Problème 1 : Colonne manquante dans certains datasets
    if len(datasets_sans_colonne) > 0 and len(datasets_avec_colonne) > 0:
        problemes_trouves = True
        print(f"\n❌ Colonne '{col}':")
        print(f"   ✓ Présente dans: {', '.join(datasets_avec_colonne)}")
        print(f"   ✗ ABSENTE dans: {', '.join(datasets_sans_colonne)}")
    
    # Problème 2 : Types de données différents
    elif len(datasets_avec_colonne) == len(noms_datasets):
        # La colonne existe partout, vérifier les types
        types_par_dataset = {}
        for nom in datasets_avec_colonne:
            types_par_dataset[nom] = str(datasets_a_comparer[nom][col].dtype)
        
        # Vérifier si tous les types sont identiques
        types_uniques = set(types_par_dataset.values())
        if len(types_uniques) > 1:
            problemes_trouves = True
            print(f"\n⚠️  Colonne '{col}' - FORMATS DIFFÉRENTS:")
            for nom, dtype in types_par_dataset.items():
                print(f"   {nom}: {dtype}")

if not problemes_trouves:
    print("\n✅ Aucun problème de compatibilité détecté ! Tous les datasets sont compatibles.")

# 5. Statistiques globales
print("\n" + "="*80)
print("STATISTIQUES GLOBALES")
print("="*80)

# Colonnes communes à tous les datasets
colonnes_communes_toutes = set.intersection(*colonnes_par_dataset.values()) if colonnes_par_dataset else set()
print(f"\nColonnes communes à TOUS les datasets: {len(colonnes_communes_toutes)}")
if len(colonnes_communes_toutes) > 0:
    print(f"Liste: {sorted(colonnes_communes_toutes)}")

# Colonnes uniques par dataset
print("\nColonnes uniques par dataset:")
for nom, colonnes in colonnes_par_dataset.items():
    autres_colonnes = set()
    for autre_nom, autres_cols in colonnes_par_dataset.items():
        if autre_nom != nom:
            autres_colonnes.update(autres_cols)
    uniques = colonnes - autres_colonnes
    if len(uniques) > 0:
        print(f"  {nom}: {len(uniques)} colonnes uniques")
        print(f"    → {sorted(uniques)}")
    else:
        print(f"  {nom}: Aucune colonne unique")

# 6. Résumé de compatibilité
print("\n" + "="*80)
print("RÉSUMÉ DE COMPATIBILITÉ")
print("="*80)
if len(noms_datasets) >= 2:
    total_colonnes = max(len(df.columns) for df in datasets_a_comparer.values())
    taux_compatibilite = (len(colonnes_communes_toutes) / total_colonnes * 100) if total_colonnes > 0 else 0
    print(f"\nNombre total de colonnes communes: {len(colonnes_communes_toutes)}")
    print(f"Taux de compatibilité global: {taux_compatibilite:.2f}%")
    print(f"Fusion possible: {'OUI ✓' if len(colonnes_communes_toutes) > 0 else 'NON ✗'}")
    
    if not problemes_trouves:
        print("\n✅ Tous les datasets ont la même structure et les mêmes types de données!")
    else:
        print(f"\n⚠️  {sum(1 for _ in toutes_les_colonnes)} problème(s) de compatibilité détecté(s) (voir ci-dessus)")
